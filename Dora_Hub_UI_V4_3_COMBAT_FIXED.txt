local PLACE_SEA_2 = 4442272183
local PLACE_SEA_3 = 7449423635

local function GetSea()
    if game.PlaceId == PLACE_SEA_1 then
        return 1
    elseif game.PlaceId == PLACE_SEA_2 then
        return 2
    elseif game.PlaceId == PLACE_SEA_3 then
        return 3
    end

    local data = LocalPlayer:FindFirstChild("Data")
    local levelObj = data and data:FindFirstChild("Level")
    local lv = levelObj and tonumber(levelObj.Value) or 1

    if lv < 700 then
        return 1
    elseif lv < 1500 then
        return 2
    end

    return 3
end

local function QuestLevelAllowedForCurrentSea(levelReq)
    local sea = GetSea()

    if sea == 1 then
        return levelReq >= 1 and levelReq < 700
    elseif sea == 2 then
        return levelReq >= 700 and levelReq < 1500
    else
        return levelReq >= 1500
    end
end

local function GetQuestCFrame(info)
    if not info then
        return nil
    end

    local key =
        tostring(info.QuestName)
        .. ":"
        .. tostring(info.QuestId)

    return QUEST_POS_BY_ID[key]
        or QUEST_POS[info.QuestName]
end

local function SetState(state)
    FARM.CurrentState = state
end

local function StripLevel(name)
    name = tostring(name or "")
    name = name:gsub(" %[%s*Lv%.%s*%d+%s*%]", "")
    name = name:gsub(" %[%s*Boss%s*%]", "")
    name = name:gsub(" %[%s*Raid Boss%s*%]", "")
    return name
end

local function GetCharacter()
    local character = LocalPlayer.Character

    if not character or not character.Parent then
        character = LocalPlayer.CharacterAdded:Wait()
    end

    local humanoid =
        character:FindFirstChildOfClass("Humanoid")
        or character:WaitForChild("Humanoid", 5)

    local root =
        character:FindFirstChild("HumanoidRootPart")
        or character:WaitForChild("HumanoidRootPart", 5)

    return character, humanoid, root
end

local function GetLevel()
    local data = LocalPlayer:FindFirstChild("Data")
    local level = data and data:FindFirstChild("Level")

    if level then
        return tonumber(level.Value) or 1
    end

    return 1
end

local function CancelMove()
    if CurrentTween then
        pcall(function()
            CurrentTween:Cancel()
        end)

        CurrentTween = nil
    end
end

local function DistanceTo(cf)
    local _, _, root = GetCharacter()

    if not root or not cf then
        return math.huge
    end

    return (root.Position - cf.Position).Magnitude
end

local function MoveToCFrame(cf, generation)
    if not cf or not FARM.Enabled then
        return false
    end

    local character, humanoid, root = GetCharacter()

    if not character
        or not humanoid
        or not root
        or humanoid.Health <= 0
    then
        return false
    end

    CancelMove()

    local distance = (root.Position - cf.Position).Magnitude

    -- Gần thì set CFrame luôn, xa thì "bay" bằng Tween.
    if distance <= 180 then
        pcall(function()
            root.CFrame = cf
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)

        return true
    end

    local duration = math.max(distance / FARM.TweenSpeed, 0.15)

    local tween = TweenServiceFarm:Create(
        root,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        ),
        {CFrame = cf}
    )

    CurrentTween = tween
    tween:Play()

    local started = tick()

    while FARM.Enabled
        and generation == FarmGeneration
        and CurrentTween == tween
        and tick() - started <= duration + 1
    do
        if not root.Parent or humanoid.Health <= 0 then
            break
        end

        if (root.Position - cf.Position).Magnitude <= 8 then
            break
        end

        task.wait(0.05)
    end

    if CurrentTween == tween then
        pcall(function()
            tween:Cancel()
        end)

        CurrentTween = nil
    end

    if FARM.Enabled and generation == FarmGeneration and root.Parent then
        pcall(function()
            root.CFrame = cf
        end)

        return true
    end

    return false
end

local QuestModule

local function GetQuestModule()
    if QuestModule then
        return QuestModule
    end

    local module = ReplicatedStorage:FindFirstChild("Quests")

    if not module or not module:IsA("ModuleScript") then
        return nil
    end

    local ok, result = pcall(require, module)

    if ok and type(result) == "table" then
        QuestModule = result
        return QuestModule
    end

    return nil
end

local function ReadLevelReq(entry)
    if type(entry) ~= "table" then
        return nil
    end

    for _, key in ipairs({
        "LevelReq",
        "Level",
        "MinLevel",
        "RequiredLevel"
    }) do
        if type(entry[key]) == "number" then
            return entry[key]
        end
    end

    return nil
end

local function ReadTask(entry)
    if type(entry) ~= "table" then
        return nil
    end

    for _, key in ipairs({
        "Task",
        "Tasks",
        "Objective",
        "Objectives"
    }) do
        if type(entry[key]) == "table" then
            return entry[key]
        end
    end

    return nil
end

local function ResolveBestQuest()
    local quests = GetQuestModule()
    local level = GetLevel()

    if not quests then
        return nil
    end

    local best = nil
    local bestLevel = -1

    for questName, entries in pairs(quests) do
        if type(entries) == "table"
            and not USELESS_QUESTS[tostring(questName)]
        then
            for questId, entry in pairs(entries) do
                if type(entry) == "table" then
                    local levelReq = ReadLevelReq(entry)
                    local taskData = ReadTask(entry)

                    if levelReq
                        and level >= levelReq
                        and QuestLevelAllowedForCurrentSea(levelReq)
                        and levelReq >= bestLevel
                        and type(taskData) == "table"
                    then
                        for mobName, count in pairs(taskData) do
                            local amount = tonumber(count)

                            if type(mobName) == "string"
                                and amount
                                and amount > 1
                            then
                                bestLevel = levelReq

                                best = {
                                    QuestName = tostring(questName),
                                    QuestId = tonumber(questId) or questId,
                                    MobName = StripLevel(mobName),
                                    LevelReq = levelReq,
                                    Sea = GetSea()
                                }
                            end
                        end
                    end
                end
            end
        end
    end

    CachedQuest = best
    return best
end

--==========================================================
-- QUEST GIVER FALLBACK
-- Nếu quest mới chưa có trong QUEST_POS, thử tìm NPC QUEST đang loaded.
--==========================================================

local function CollectQuestNPCs()
    local list = {}
    local seen = {}

    local function AddFromObject(obj)
        if not obj or seen[obj] then
            return
        end

        if obj.Name ~= "QuestBBG" then
            return
        end

        local parent = obj.Parent
        local model = parent

        while model and not model:IsA("Model") do
            model = model.Parent
        end

        if not model then
            return
        end

        local head =
            model:FindFirstChild("Head")
            or model:FindFirstChild("HumanoidRootPart")
            or model.PrimaryPart

        if head and head:IsA("BasePart") then
            seen[obj] = true

            table.insert(
                list,
                head.CFrame * CFrame.new(0, -2, 2)
            )
        end
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        AddFromObject(obj)
    end

    if type(getnilinstances) == "function" then
        pcall(function()
            for _, obj in ipairs(getnilinstances()) do
                AddFromObject(obj)
            end
        end)
    end

    return list
end

local function TryAcceptQuestAtLoadedNPCs(info, generation)
    if not CommF or not info then
        return false
    end

    local npcs = CollectQuestNPCs()

    for _, cf in ipairs(npcs) do
        if not FARM.Enabled or generation ~= FarmGeneration then
            return false
        end

        MoveToCFrame(cf, generation)

        if DistanceTo(cf) <= 12 then
            pcall(function()
                CommF:InvokeServer(
                    "StartQuest",
                    info.QuestName,
                    info.QuestId
                )
            end)

            task.wait(0.7)

            if QuestVisible() then
                local title = StripLevel(QuestTitle())

                if title:lower():find(
                    info.MobName:lower(),
                    1,
                    true
                ) then
                    QUEST_POS[info.QuestName] = cf
                    return true
                end

                pcall(function()
                    CommF:InvokeServer("AbandonQuest")
                end)
            end
        end
    end

    return false
end

local function EnsureCorrectQuest(info, generation)
    if not FARM.AutoQuest then
        return true
    end

    if not info or not CommF then
        return false
    end

    if QuestVisible() then
        local title = StripLevel(QuestTitle()):lower()
        local mob = info.MobName:lower()

        if title:find(mob, 1, true) then
            return true
        end

        SetState("Abandon wrong quest")

        pcall(function()
            CommF:InvokeServer("AbandonQuest")
        end)

        task.wait(0.5)
    end

    SetState("Flying to quest")

    local questCF = GetQuestCFrame(info)

    if questCF then
        MoveToCFrame(questCF, generation)

        if not FARM.Enabled or generation ~= FarmGeneration then
            return false
        end

        if DistanceTo(questCF) <= 15 then
            SetState("Taking quest")

            pcall(function()
                CommF:InvokeServer("SetSpawnPoint")
            end)

            task.wait(0.15)

            pcall(function()
                CommF:InvokeServer(
                    "StartQuest",
                    info.QuestName,
                    info.QuestId
                )
            end)

            task.wait(0.9)

            if QuestVisible() then
                return true
            end
        end
    end

    -- Fallback cho quest mới/map mới.
    SetState("Finding quest NPC")
    return TryAcceptQuestAtLoadedNPCs(info, generation)
end

local function GetEnemyParts(enemy)
    if not enemy or not enemy:IsA("Model") then
        return nil, nil
    end

    if Players:GetPlayerFromCharacter(enemy) then
        return nil, nil
    end

    local humanoid = enemy:FindFirstChildOfClass("Humanoid")
    local root =
        enemy:FindFirstChild("HumanoidRootPart")
        or enemy:FindFirstChild("UpperTorso")
        or enemy:FindFirstChild("Torso")
        or enemy.PrimaryPart

    if not humanoid or humanoid.Health <= 0 or not root then
        return nil, nil
    end

    return humanoid, root
end

local function NameMatchesMob(modelName, mobName)
    return StripLevel(modelName):lower()
        == StripLevel(mobName):lower()
end

local function FindQuestEnemy(mobName)
    local _, _, myRoot = GetCharacter()

    if not myRoot then
        return nil
    end

    local enemies = workspace:FindFirstChild("Enemies")
    local nearest
    local nearestDistance = math.huge

    if enemies then
        for _, enemy in ipairs(enemies:GetChildren()) do
            if NameMatchesMob(enemy.Name, mobName) then
                local hum, root = GetEnemyParts(enemy)

                if hum and root then
                    local distance =
                        (myRoot.Position - root.Position).Magnitude

                    if distance < nearestDistance then
                        nearestDistance = distance
                        nearest = enemy
                    end
                end
            end
        end
    end

    return nearest
end

local function FindMobSpawnCFrame(mobName)
    local worldOrigin = workspace:FindFirstChild("_WorldOrigin")
    local spawns =
        worldOrigin and worldOrigin:FindFirstChild("EnemySpawns")

    if not spawns then
        return nil
    end

    local clean = StripLevel(mobName):lower()
    local best

    for _, spawnObj in ipairs(spawns:GetChildren()) do
        if StripLevel(spawnObj.Name):lower() == clean then
            if spawnObj:IsA("BasePart") then
                best = spawnObj.CFrame
                break
            elseif spawnObj:IsA("Model") then
                local part =
                    spawnObj.PrimaryPart
                    or spawnObj:FindFirstChildWhichIsA("BasePart")

                if part then
                    best = part.CFrame
                    break
                end
            end
        end
    end

    return best
end


local function FindStoredMobCFrame(mobName)
    local clean = StripLevel(mobName):lower()

    for _, obj in ipairs(ReplicatedStorage:GetChildren()) do
        if obj:IsA("Model")
            and StripLevel(obj.Name):lower() == clean
        then
            local root =
                obj:FindFirstChild("HumanoidRootPart")
                or obj.PrimaryPart
                or obj:FindFirstChildWhichIsA("BasePart")

            if root then
                return root.CFrame
            end
        end
    end

    return nil
end

local function BringSameMobs(mobName, anchorEnemy)
    if not FARM.BringMob or not anchorEnemy then
        return
    end

    local enemies = workspace:FindFirstChild("Enemies")
    if not enemies then
        return
    end

    local anchorHum, anchorRoot = GetEnemyParts(anchorEnemy)
    if not anchorHum or not anchorRoot then
        return
    end

    TryExpandSimulationRadius()

    local cleanName = StripLevel(mobName):lower()
    local anchorCF =
        CFrame.new(anchorRoot.Position)
        * CFrame.new(FARM.BringOffset)

    for _, enemy in ipairs(enemies:GetChildren()) do
        if enemy ~= anchorEnemy
            and enemy:IsA("Model")
            and StripLevel(enemy.Name):lower() == cleanName
        then
            local hum, root = GetEnemyParts(enemy)

            if hum and root and hum.Health > 0 then
                local distance = (root.Position - anchorRoot.Position).Magnitude

                if distance <= FARM.BringRadius then
                    pcall(function()
                        root.CanCollide = false
                        root.Size = FARM.HitboxSize
                        root.CFrame = anchorCF
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero

                        hum.AutoRotate = false
                    end)
                end
            end
        end
    end
end

local function ToolScore(tool)
    if not tool or not tool:IsA("Tool") then
        return -math.huge
    end

    local tooltip = string.lower(tostring(tool.ToolTip or ""))
    local name = string.lower(tool.Name)

    if tooltip:find("melee", 1, true) then
        return 100
    end

    if tooltip:find("sword", 1, true) then
        return 90
    end

    local meleeNames = {
        "combat",
        "godhuman",
        "superhuman",
        "sharkman",
        "death step",
        "electric",
        "dragon",
        "sanguine",
        "water kung fu",
        "dark step"
    }

    for _, key in ipairs(meleeNames) do
        if name:find(key, 1, true) then
            return 85
        end
    end

    if tooltip:find("gun", 1, true) then
        return 25
    end

    if tooltip:find("fruit", 1, true)
        or name:find("fruit", 1, true)
    then
        return 10
    end

    return 50
end

local function FindBestWeapon(character)
    local bestTool
    local bestScore = -math.huge

    local function Check(container)
        if not container then
            return
        end

        for _, item in ipairs(container:GetChildren()) do
            if item:IsA("Tool") then
                local score = ToolScore(item)

                if score > bestScore then
                    bestScore = score
                    bestTool = item
                end
            end
        end
    end

    Check(character)
    Check(LocalPlayer:FindFirstChildOfClass("Backpack"))

    return bestTool
end

local function EquipBestWeapon(character, humanoid)
    local tool = FindBestWeapon(character)

    if not tool then
        return nil
    end

    if tool.Parent ~= character then
        pcall(function()
            humanoid:EquipTool(tool)
        end)

        task.wait(0.03)
    end

    if tool.Parent == character then
        return tool
    end

    return character:FindFirstChildOfClass("Tool")
end

local function PrepareEnemy(enemy)
    local hum, root = GetEnemyParts(enemy)

    if not hum or not root then
        return nil, nil
    end

    pcall(function()
        root.CanCollide = false
        root.Size = FARM.HitboxSize
        root.Transparency = math.max(root.Transparency, 0.65)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)

    return hum, root
end

local function MoveAboveEnemy(enemy)
    local _, humanoid, myRoot = GetCharacter()
    local enemyHumanoid, enemyRoot = PrepareEnemy(enemy)

    if not humanoid
        or not myRoot
        or humanoid.Health <= 0
        or not enemyHumanoid
        or not enemyRoot
    then
        return false
    end

    CancelMove()

    local cf =
        enemyRoot.CFrame
        * CFrame.new(0, FARM.Height, FARM.Behind)

    pcall(function()
        myRoot.CFrame = cf
        myRoot.AssemblyLinearVelocity = Vector3.zero
        myRoot.AssemblyAngularVelocity = Vector3.zero
    end)

    return true
end

local function FireToolActivated(tool)
    if not tool or not tool:IsA("Tool") then
        return false
    end

    local fired = false

    -- Cách chuẩn của Roblox.
    pcall(function()
        if tool.Enabled == false then
            tool.Enabled = true
        end

        tool:Activate()
        fired = true
    end)

    -- Không giả lập chuột/touch.
    -- firesignal kích hoạt signal Activated trực tiếp nếu executor hỗ trợ.
    if type(firesignal) == "function" then
        pcall(function()
            firesignal(tool.Activated)
            fired = true
        end)
    end

    -- Fallback cho executor hỗ trợ getconnections.
    if type(getconnections) == "function" then
        pcall(function()
            for _, connection in ipairs(getconnections(tool.Activated)) do
                if connection then
                    if type(connection.Fire) == "function" then
                        pcall(function()
                            connection:Fire()
                            fired = true
                        end)
                    elseif type(connection.Function) == "function" then
                        pcall(function()
                            connection.Function()
                            fired = true
                        end)
                    end
                end
            end
        end)
    end

    return fired
end

local function AttackOnce()
    local character, humanoid = GetCharacter()

    if not character
        or not humanoid
        or humanoid.Health <= 0
    then
        return false
    end

    local tool = EquipBestWeapon(character, humanoid)

    if not tool or tool.Parent ~= character then
        return false
    end

    -- Không dùng VirtualUser / VirtualInputManager.
    -- Vì vậy farm đánh được mà không chiếm thao tác màn hình/UI.
    return FireToolActivated(tool)
end

local function SetFarmNoclip(enabled)
    if NoclipConnection then
        NoclipConnection:Disconnect()
        NoclipConnection = nil
    end

    if not enabled then
        return
    end

    NoclipConnection = RunService.Stepped:Connect(function()
        if not FARM.Enabled then
            return
        end

        local character = LocalPlayer.Character

        if not character then
            return
        end

        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end)
end

--==========================================================
-- STATE MACHINE
--
-- 1. Đọc level + quest phù hợp
-- 2. Bay tới NPC nhận quest
-- 3. Kiểm tra quest title
-- 4. Tìm đúng mob quest
-- 5. Bay tới mob
-- 6. Chỉ lúc này mới đánh
--==========================================================

local function StartAutoFarm()
    FarmGeneration += 1
    local generation = FarmGeneration

    SetFarmNoclip(true)

    task.spawn(function()
        while FARM.Enabled and generation == FarmGeneration do
            local _, humanoid, root = GetCharacter()

            if not humanoid or not root or humanoid.Health <= 0 then
                SetState("Waiting respawn")
                task.wait(1)
                continue
            end

            local info = ResolveBestQuest()

            if not info then
                SetState("Quest data unavailable")
                task.wait(1)
                continue
            end

            -- =================================================
            -- STEP 1: NHẬN QUEST
            -- =================================================

            if FARM.AutoQuest then
                local hasQuest = EnsureCorrectQuest(info, generation)

                if not hasQuest then
                    SetState("Retry quest")
                    task.wait(0.5)
                    continue
                end
            end

            if not FARM.Enabled or generation ~= FarmGeneration then
                break
            end

            -- =================================================
            -- STEP 2: ĐÚNG QUEST -> TÌM ĐÚNG MOB
            -- =================================================

            if FARM.AutoQuest and not QuestVisible() then
                CurrentTarget = nil
                task.wait(0.15)
                continue
            end

            if FARM.AutoQuest then
                local title = StripLevel(QuestTitle()):lower()

                if not title:find(
                    info.MobName:lower(),
                    1,
                    true
                ) then
                    CurrentTarget = nil

                    pcall(function()
                        CommF:InvokeServer("AbandonQuest")
                    end)

                    task.wait(0.5)
                    continue
                end
            end

            local target = FindQuestEnemy(info.MobName)

            if not target then
                local spawnCF =
                    FindMobSpawnCFrame(info.MobName)
                    or FindStoredMobCFrame(info.MobName)

                if spawnCF then
                    SetState("Flying to " .. info.MobName)

                    MoveToCFrame(
                        spawnCF * CFrame.new(0, 15, 0),
                        generation
                    )
                else
                    SetState("Waiting target: " .. info.MobName)
                    task.wait(0.35)
                end

                CurrentTarget = nil
                task.wait(0.15)
                continue
            end

            CurrentTarget = target
            SetState("Farming " .. info.MobName)

            -- =================================================
            -- STEP 3: COMBAT
            -- =================================================

            while FARM.Enabled
                and generation == FarmGeneration
                and CurrentTarget
            do
                local enemyHumanoid =
                    CurrentTarget:FindFirstChildOfClass("Humanoid")

                if not enemyHumanoid
                    or enemyHumanoid.Health <= 0
                    or not CurrentTarget.Parent
                then
                    break
                end

                -- Quest hoàn thành/mất thì NGỪNG đánh và quay lại nhận quest.
                if FARM.AutoQuest and not QuestVisible() then
                    break
                end

                if FARM.AutoQuest then
                    local title =
                        StripLevel(QuestTitle()):lower()

                    if not title:find(
                        info.MobName:lower(),
                        1,
                        true
                    ) then
                        break
                    end
                end

                BringSameMobs(info.MobName, CurrentTarget)

                if not MoveAboveEnemy(CurrentTarget) then
                    break
                end

                local attacked = AttackOnce()

                if FARM.FastAttack and attacked then
                    task.wait(0.02)
                    AttackOnce()
                end

                task.wait(
                    FARM.FastAttack
                    and FARM.FastAttackDelay
                    or FARM.NormalAttackDelay
                )
            end

            CurrentTarget = nil
            task.wait(0.08)
        end

        CancelMove()
        CurrentTarget = nil
        SetState("Idle")
    end)
end

local function SetAutoFarm(enabled)
    enabled = enabled == true

    if FARM.Enabled == enabled then
        return
    end

    FARM.Enabled = enabled

    if enabled then
        CachedQuest = nil
        CurrentTarget = nil
        StartAutoFarm()
    else
        FarmGeneration += 1
        CurrentTarget = nil
        CachedQuest = nil
        CancelMove()
        SetFarmNoclip(false)
        SetState("Idle")
    end
end

            Position = state and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9)
        })
        if callback then
            task.spawn(function()
                callback(state)

        end
    end

    -- (MouseButton1Click + holder.InputBegan), nguyên nhân làm ON -> OFF ngay.

Stroke(Floating, Color3.fromRGB(120, 210, 255), 2, 0)

Minimize.MouseButton1Click:Connect(function()
    Tween(Main, 0.25, {Size = UDim2.new(0, 0, 0, 0)})
    task.wait(0.2)
    Main.Visible = false
    Floating.Visible = true
    Floating.Size = UDim2.new(0, 0, 0, 0)
    Tween(Floating, 0.25, {Size = UDim2.new(0, 56, 0, 56)})
end)

Floating.MouseButton1Click:Connect(function()
    Tween(Floating, 0.2, {Size = UDim2.new(0, 0, 0, 0)})
    task.wait(0.15)
    Floating.Visible = false
    Main.Visible = true
    Main.Size = UDim2.new(0, 0, 0, 0)
    Tween(Main, 0.3, {Size = UDim2.new(0, 550, 0, 350)})
end)

Close.MouseButton1Click:Connect(function()
    FARM.Enabled = false
    FarmGeneration += 1
    CurrentTarget = nil
    CancelMove()
    SetFarmNoclip(false)

    Tween(Main, 0.25, {Size = UDim2.new(0, 0, 0, 0)})
    task.wait(0.25)
    ScreenGui:Destroy()
end)

do
    local dragging = false
    local dragInput
    local dragStart
    local startPosition

    TopBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then

            dragging = true
            dragStart = input.Position
            startPosition = Main.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    TopBar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)
end

do
    local dragging = false
    local startPos
    local dragStart

    Floating.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch
            or input.UserInputType == Enum.UserInputType.MouseButton1 then

            dragging = true
            dragStart = input.Position
            startPos = Floating.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end

        if input.UserInputType == Enum.UserInputType.Touch
            or input.UserInputType == Enum.UserInputType.MouseMovement then

            local delta = input.Position - dragStart
            Floating.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch
            or input.UserInputType == Enum.UserInputType.MouseButton1 then

            dragging = false
        end
    end)
end

local finalSize = Main.Size
Main.Size = UDim2.new(0, 0, 0, 0)
Tween(Main, 0.45, {Size = finalSize})

task.wait(0.5)

Notify("Dora Hub", "Doraemon Hub Loaded!", 4)

print("[Dora Hub] Loaded")
