local FruitState = {
    -- Busy chỉ dành cho thao tác CÓ DI CHUYỂN (pickup),
    -- để Farm biết lúc nào phải nhường quyền điều khiển nhân vật.

    RandomBusy = false,
    StoreBusy = false,
    PickupBusy = false,

    -- Chặn Auto Store vài giây sau Random để thấy fruit thật.
    StoreBlockUntil = 0,

    -- Theo dõi fruit thật sự vừa vào Backpack.
    AcquiredSerial = 0,
    LastAcquiredName = nil,

    RandomInterval = 1.0,
    StoreInterval = 1.2,
    PickupInterval = 0.45,

    LastRandom = 0,
    LastStore = 0,
    LastPickup = 0,

    FruitCatalog = nil,
    FruitCatalogAt = 0,

    StoredTools = setmetatable({}, {__mode = "k"}),
}

local function getFruitCommRemote()
    local remotes =
        ReplicatedStorage:FindFirstChild("Remotes")
        or ReplicatedStorage:WaitForChild("Remotes", 8)

    if not remotes then
        return nil
    end

    -- Current Blox Fruits remote.
    return remotes:FindFirstChild("CommF_")
        or remotes:WaitForChild("CommF_", 8)
end

local function invokeCommF(...)
    local remote = getFruitCommRemote()

    if not remote then
        return false, "COMM_F_NOT_FOUND"
    end

    local args = table.pack(...)

    local ok, result = pcall(function()
        return remote:InvokeServer(
            table.unpack(args, 1, args.n)
        )
    end)

    if not ok then
        return false, tostring(result)
    end

    return true, result
end

local function normalizeFruitToken(value)
    value = string.lower(tostring(value or ""))

    value = value:gsub("%s+[Ff][Rr][Uu][Ii][Tt]$", "")
    value = value:gsub("fruit$", "")
    value = value:gsub("[^%w]", "")

    return value
end

local function looksLikeFruitTool(tool)
    if not tool or not tool:IsA("Tool") then
        return false
    end

    -- Đây là dấu hiệu đáng tin cậy hơn trong Blox Fruits hiện tại.
    -- Fruit Tool có child/object tên "Fruit".
    if tool:FindFirstChild("Fruit") then
        return true
    end

    local lowerName = string.lower(tool.Name)

    -- Fallback cho executor/client load chậm child Fruit.
    if lowerName:find("fruit", 1, true) then
        return true
    end

    local okAttr, isFruit = pcall(function()
        return tool:GetAttribute("Fruit")
            or tool:GetAttribute("BloxFruit")
    end)

    if okAttr and isFruit then
        return true
    end

    return false
end

local function getFruitCatalog(force)
    local now = os.clock()

    if not force
        and FruitState.FruitCatalog
        and now - FruitState.FruitCatalogAt < 90 then

        return FruitState.FruitCatalog
    end

    local remote = getFruitCommRemote()
    if not remote then
        return FruitState.FruitCatalog
    end

    local ok, result = pcall(function()
        return remote:InvokeServer("GetFruits")
    end)

    if ok and type(result) == "table" then
        FruitState.FruitCatalog = result
        FruitState.FruitCatalogAt = now
    end

    return FruitState.FruitCatalog
end

local function resolveStoredFruitName(tool)
    if not tool then
        return nil
    end

    local toolToken = normalizeFruitToken(tool.Name)

    if toolToken == "" then
        return nil
    end

    local catalog = getFruitCatalog(false)

    if type(catalog) == "table" then
        local bestName = nil
        local bestScore = -1

        for _, fruit in pairs(catalog) do
            local internalName = nil

            if type(fruit) == "table" then
                internalName =
                    fruit.Name
                    or fruit.name
                    or fruit.FruitName
            elseif type(fruit) == "string" then
                internalName = fruit
            end

            if internalName then
                local internalToken =
                    normalizeFruitToken(internalName)

                local score = -1

                if internalToken == toolToken then
                    score = 1000
                elseif internalToken:find(toolToken, 1, true) then
                    score = #toolToken
                elseif toolToken:find(internalToken, 1, true) then
                    score = #internalToken
                end

                if score > bestScore then
                    bestScore = score
                    bestName = internalName
                end
            end
        end

        if bestName then
            return bestName
        end
    end

    -- Fallback cho tên "Rocket Fruit" -> "Rocket-Rocket".
    local display = tool.Name:gsub("%s+[Ff][Rr][Uu][Ii][Tt]$", "")
    display = display:gsub("^%s+", ""):gsub("%s+$", "")

    if display ~= "" then
        return display .. "-" .. display
    end

    return nil
end

local function getOwnedFruitTools()
    local result = {}

    local backpack = Player:FindFirstChild("Backpack")
    local character = Player.Character

    local function scan(container)
        if not container then
            return
        end

        for _, obj in ipairs(container:GetChildren()) do
            if looksLikeFruitTool(obj) then
                table.insert(result, obj)
            end
        end
    end

    scan(backpack)
    scan(character)

    return result
end

local function storeFruitTool(tool)
    if not tool
        or not tool.Parent
        or FruitState.StoredTools[tool] then

        return false, "SKIP"
    end

    local remote = getFruitCommRemote()
    if not remote then
        return false, "NO_COMMF"
    end

    local internalName = resolveStoredFruitName(tool)

    if not internalName then
        return false, "UNKNOWN_FRUIT_NAME"
    end

    local ok, result = pcall(function()
        return remote:InvokeServer(
            "StoreFruit",
            internalName,
            tool
        )
    end)

    if ok and result == true then
        FruitState.StoredTools[tool] = true
        return true, internalName
    end

    if ok then
        return false,
            "STORE_SERVER: " ..
            tostring(result)
    end

    return false, tostring(result)
end

local function storeAllOwnedFruits()
    if FruitState.StoreBusy then
        return 0
    end

    FruitState.StoreBusy = true

    local stored = 0
    local tools = getOwnedFruitTools()

    for _, tool in ipairs(tools) do
        local ok = storeFruitTool(tool)

        if ok then
            stored += 1
        end

        task.wait(0.08)
    end

    FruitState.StoreBusy = false
    return stored
end

local function findWorldFruitTool()
    local root = Player.Character
        and Player.Character:FindFirstChild("HumanoidRootPart")

    local nearest = nil
    local nearestHandle = nil
    local nearestDistance = math.huge

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Tool")
            and looksLikeFruitTool(obj) then

            -- Không lấy fruit đã nằm trong player inventory.
            if not obj:IsDescendantOf(Player) then
                local handle =
                    obj:FindFirstChild("Handle")
                    or obj:FindFirstChildWhichIsA("BasePart", true)

                if handle and handle:IsA("BasePart") then
                    local distance = 0

                    if root then
                        distance =
                            (handle.Position - root.Position).Magnitude
                    end

                    if distance < nearestDistance then
                        nearest = obj
                        nearestHandle = handle
                        nearestDistance = distance
                    end
                end
            end
        end
    end

    return nearest, nearestHandle, nearestDistance
end

local function pickupFruitTool(tool, handle)
    if not tool or not handle then
        return false, "NO_FRUIT"
    end

    local character = Player.Character
    local humanoid = character
        and character:FindFirstChildOfClass("Humanoid")
    local root = character
        and character:FindFirstChild("HumanoidRootPart")

    if not character or not humanoid or not root then
        return false, "NO_CHARACTER"
    end

    if FruitState.PickupBusy then
        return false, "PICKUP_BUSY"
    end

    FruitState.PickupBusy = true
    FruitState.Busy = true

    local oldCFrame = root.CFrame
    local picked = false

    -- Ưu tiên executor touch API nếu có.
    if type(firetouchinterest) == "function" then
        pcall(function()
            firetouchinterest(
                root,
                handle,
                0
            )

            task.wait(0.05)

            firetouchinterest(
                root,
                handle,
                1
            )

            picked = true
        end)
    end

    -- Fallback: teleport sát fruit một khoảng rất ngắn.
    if not picked then
        pcall(function()
            root.CFrame =
                handle.CFrame * CFrame.new(0, 1.5, 0)

            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)

        task.wait(0.18)
    end

    -- Chờ fruit vào Backpack.
    local timeout = os.clock() + 1.25

    while tool.Parent
        and not tool:IsDescendantOf(Player)
        and os.clock() < timeout do

        task.wait(0.05)
    end

    if tool:IsDescendantOf(Player) then
        picked = true
    end

    -- Trả về vị trí cũ nếu vẫn còn nhân vật.
    pcall(function()
        if root.Parent then
            root.CFrame = oldCFrame
        end
    end)

    FruitState.PickupBusy = false

local function ownStyle(styleName)
    local backpack =
        Player:FindFirstChild("Backpack")
    local character =
        Player.Character

    if backpack
        and backpack:FindFirstChild(styleName) then

        return true
    end

    if character
        and character:FindFirstChild(styleName) then

        return true
    end

    return false
end

local function ownStyleEntry(style)
    if not style then
        return false
    end

    return ownStyle(
        style.InternalLabel
        or style.Label
    )
end

local function invokeStyleRaw(...)
    local args = table.pack(...)

    local remote =
        ReplicatedStorage
        :WaitForChild("Remotes")
        :WaitForChild("CommF_")

    local ok, result = pcall(function()
        return remote:InvokeServer(
            table.unpack(args, 1, args.n)
        )
    end)

    return ok, result
end

local function buyFightingStyle(style)
    if not style then
        return false, "NO_STYLE"
    end

    if StyleShopState.Busy then
        return false, "BUSY"
    end

    StyleShopState.Busy = true

    local ok = false
    local r1 = nil
    local r2 = nil

    if style.Special == "DRAGON_CLAW" then
        local ok1
        ok1, r1 = invokeStyleRaw(
            "BlackbeardReward",
            "DragonClaw",
            "1"
        )

        task.wait(0.08)

        local ok2
        ok2, r2 = invokeStyleRaw(
            "BlackbeardReward",
            "DragonClaw",
            "2"
        )

        ok = ok1 and ok2
    else
        -- SOURCE SHOP hiện tại:
        -- FireRemote("BuyBlackLeg"), FireRemote("BuyElectro"), ...
        ok, r1 = invokeStyleRaw(
            style.Remote
        )
    end

    StyleShopState.Busy = false
    StyleShopState.LastBuy = os.clock()

    if not ok then
        return false,
            "REMOTE ERROR | " ..
            tostring(r1) ..
            " | " ..
            tostring(r2)
    end

    return true,
        "ĐÃ GỬI SERVER | A=" ..
        tostring(r1) ..
        " | B=" ..
        tostring(r2)
end

local function getRedeemRemote()
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")
    return remotes and remotes:FindFirstChild("Redeem")

local function redeemAllCodes(force)
    if UtilityState.RedeemRunning then
        return
    end

    if UtilityState.RedeemedThisSession and not force then
        return
    end

    UtilityState.RedeemRunning = true

    task.spawn(function()
        local remote = getRedeemRemote()

        if not remote then
            RedeemStatus.Text = "Lỗi: không tìm thấy Remotes.Redeem"
            UtilityState.RedeemRunning = false
            return
        end

        local attempted = 0

        for index, code in ipairs(ALL_REDEEM_CODES) do
            if not UtilityState.AutoRedeem and not force then
                break
            end

            RedeemStatus.Text =
                "Redeem " .. tostring(index) ..
                "/" .. tostring(#ALL_REDEEM_CODES) ..
                ": " .. tostring(code)

            local ok = pcall(function()
                remote:InvokeServer(code)
            end)

            if ok then
                attempted += 1
            end

            -- Không spam RemoteFunction.
            task.wait(0.16)
        end

        UtilityState.RedeemedThisSession = true
        UtilityState.RedeemRunning = false

        RedeemStatus.Text =
            "Hoàn tất • đã thử " ..
            tostring(attempted) ..
            "/" .. tostring(#ALL_REDEEM_CODES) ..
            " code"
    end)
end

    if UtilityState.AutoRedeem then
        redeemAllCodes(false)
    end
end)

local function belongsToAxiom(instance)
    local current = instance

    while current do
        if current == ScreenGui then
            return true
        end

        current = current.Parent
    end

    return false
end

local function isNotificationObject(instance)
    if not instance or belongsToAxiom(instance) then
        return false
    end

    local lowered = string.lower(instance.Name or "")

    if NotificationNames[lowered] then
        return true
    end

    -- Chỉ match tên rõ ràng, tránh giết Main/UI gameplay.
    if lowered:find("notification", 1, true)
        or lowered:find("notify", 1, true) then

        return true
    end

    return false
end

local function hideNotificationObject(instance)
    if not UtilityState.HideNotifications
        or not isNotificationObject(instance) then
        return
    end

    pcall(function()
        if instance:IsA("ScreenGui") then
            instance.Enabled = false

        elseif instance:IsA("GuiObject") then
            instance.Visible = false
        end
    end)
end

local function sweepNotifications()
    if not UtilityState.HideNotifications then
        return
    end

    local playerGui = Player:FindFirstChildOfClass("PlayerGui")
        or Player:FindFirstChild("PlayerGui")

    if not playerGui then
        return
    end

    for _, obj in ipairs(playerGui:GetDescendants()) do
        hideNotificationObject(obj)
    end
end

local playerGuiForNotify = Player:FindFirstChildOfClass("PlayerGui")
    or Player:FindFirstChild("PlayerGui")

    playerGuiForNotify.DescendantAdded:Connect(function(obj)
        if UtilityState.HideNotifications then
            task.defer(function()
                hideNotificationObject(obj)
            end)
        end
    end)
end

local function getStatsRemote()
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")
    return remotes and remotes:FindFirstChild("CommF_")
end

local function getAvailablePoints()
    local data = Player:FindFirstChild("Data")
    local points = data and data:FindFirstChild("Points")

    if points and tonumber(points.Value) then
        return tonumber(points.Value)
    end

    return 0
end

local function getStatLevel(statName)
    local data = Player:FindFirstChild("Data")
    local stats = data and data:FindFirstChild("Stats")
    local stat = stats and stats:FindFirstChild(statName)
    local level = stat and stat:FindFirstChild("Level")

    if level and tonumber(level.Value) then
        return tonumber(level.Value)
    end

    return 0
end

local function resolveAutoStatTarget()
    return UtilityState.SelectedStat or "Melee"
end

local function addStatPoints(statName, amount)
    local remote = getStatsRemote()

    if not remote then
        return false, "NO_COMMF"
    end

    if amount <= 0 then
        return false, "NO_POINTS"
    end

    local ok = pcall(function()
        remote:InvokeServer(
            "AddPoint",
            statName,
            amount
        )
    end)

    return ok, ok and "OK" or "REMOTE_FAILED"
end

AutoStatsToggle.MouseButton1Click:Connect(function()
    UtilityState.AutoStats = not UtilityState.AutoStats
    setUtilityToggle(AutoStatsToggle, UtilityState.AutoStats)

    if not UtilityState.AutoStats then
        StatsStatus.Text = "Trạng thái: Tắt"
    end
end)

setUtilityToggle(AutoStatsToggle, UtilityState.AutoStats)

task.spawn(function()
    while ScreenGui.Parent do
        if UtilityState.AutoStats then
            local points = getAvailablePoints()

            if points > 0 then
                local target = resolveAutoStatTarget()
                local amount = math.min(
                    UtilityState.StatBatch,
                    points
                )

                local ok, status =
                    addStatPoints(target, amount)

                if ok then
                    StatsStatus.Text =
                        "Đang +" .. tostring(amount) ..
                        " → " .. target ..
                        " | Còn: " ..
                        tostring(getAvailablePoints())
                else
                    StatsStatus.Text =
                        "Stat lỗi: " .. tostring(status)
                end

                task.wait(UtilityState.StatDelay)
            else
                StatsStatus.Text =
                    "Đang bật • chưa có điểm stat"
                task.wait(0.45)
            end
        else
            task.wait(0.35)
        end
    end
end)

end

local TravelState = {
    Tween = nil,
    Active = false,
    MovementLock = false,
    TargetKey = nil,
    TargetPosition = nil,
    Token = 0,

    -- Giữ thông tin lần vừa tới bãi để Farm không lập tức
    -- quay về NPC quest khi GUI/streaming đang load.
    ArrivedAt = 0,
    LastArrivalKey = nil,
    ArrivalGrace = 4.0,

    -- Chậm + chia đoạn để giảm rubber-band khi đi xa.
    Speed = 225,
    VerticalSpeed = 250,
    SegmentLength = 180,

    SafeBaseY = 220,
    SafeClearance = 135,
    ArrivalHold = 0.55,

    Humanoid = nil,
}

local function cancelTravel()
    TravelState.Token += 1

    if TravelState.Tween then
        pcall(function()
            TravelState.Tween:Cancel()
        end)
    end

    TravelState.Tween = nil
    TravelState.Active = false
    TravelState.MovementLock = false
    TravelState.TargetKey = nil
    TravelState.TargetPosition = nil

    pcall(function()
        if TravelState.Humanoid
            and TravelState.Humanoid.Parent then

            TravelState.Humanoid.AutoRotate = true
            TravelState.Humanoid.Sit = false
        end
    end)

    TravelState.Humanoid = nil
end

local function flatFacingCFrame(position, sourceCFrame)
    local look = sourceCFrame.LookVector
    local flat = Vector3.new(look.X, 0, look.Z)

    if flat.Magnitude < 0.01 then
        flat = Vector3.new(0, 0, -1)
    else
        flat = flat.Unit
    end

    return CFrame.lookAt(
        position,
        position + flat
    )
end

local function tweenTravelPhase(
    root,
    targetCFrame,
    speed,
    token
)
    if not root
        or not root.Parent
        or token ~= TravelState.Token then
        return false
    end

    local distance =
        (root.Position - targetCFrame.Position).Magnitude

    if distance <= 2 then
        root.CFrame = targetCFrame
        return true
    end

    local duration = math.clamp(
        distance / speed,
        0.12,
        8
    )

    local tween = TweenService:Create(
        root,
        TweenInfo.new(
            duration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        ),
        {CFrame = targetCFrame}
    )

    TravelState.Tween = tween
    tween:Play()

    local playbackState = tween.Completed:Wait()

    if token ~= TravelState.Token then
        return false
    end

    return playbackState == Enum.PlaybackState.Completed
end

local function travelHorizontalSegments(
    root,
    destination,
    safeY,
    token
)
    local from = Vector3.new(
        root.Position.X,
        safeY,
        root.Position.Z
    )

    local to = Vector3.new(
        destination.X,
        safeY,
        destination.Z
    )

    local delta = to - from
    local distance = delta.Magnitude

    if distance <= 2 then
        return true
    end

    local segments = math.max(
        1,
        math.ceil(distance / TravelState.SegmentLength)
    )

    for i = 1, segments do
        if token ~= TravelState.Token or not root.Parent then
            return false
        end

        local p = from:Lerp(to, i / segments)
        local ok = tweenTravelPhase(
            root,
            flatFacingCFrame(p, root.CFrame),
            TravelState.Speed,
            token
        )

        if not ok then
            return false
        end

        -- nhường 1 nhịp nhỏ để vị trí được đồng bộ trước đoạn kế.
        task.wait(0.035)
    end

    return true
end

        elseif mob == "Shanda"

            or mob == "Royal Soldier" then

            remote:InvokeServer(
                "requestEntrance",
                Vector3.new(-7894.6177, 5547.1416, -380.2912)

        elseif mob == "Ship Deckhand"
            or mob == "Ship Engineer"

            or mob == "Ship Officer" then

            remote:InvokeServer(
                "requestEntrance",

        end
    end)
end

local function nameMatchesMob(modelName, wantedName)
    if not modelName or not wantedName then
        return false
    end

    return string.find(
        string.lower(modelName),
        string.lower(wantedName),
        1,
        true
    ) ~= nil
end

local function findReplicatedMobRoot(wantedName)
    if not wantedName then
        return nil
    end

    -- Current scripts normally look at top-level ReplicatedStorage
    -- only as presence/proxy fallback.
    for _, obj in ipairs(ReplicatedStorage:GetChildren()) do
        if obj:IsA("Model")
            and nameMatchesMob(
                obj.Name,
                wantedName
            ) then

            local root =
                obj:FindFirstChild("HumanoidRootPart")
                or obj:FindFirstChild("Torso")
                or obj.PrimaryPart

            if root and root:IsA("BasePart") then
                return root
            end
        end
    end

    return nil
end

local function getMobAreaCFrame(q, fromPosition)
    if not q then
        return nil
    end

    -- PRIMARY: known CFrameMon/static spawn.
    local staticArea =
        MOB_SPAWNS[q.Mob]

    if staticArea then
        return staticArea
    end

    -- FALLBACK only when static data is missing.
    local proxyRoot =
        findReplicatedMobRoot(q.Mob)

    return proxyRoot and proxyRoot.CFrame
        or nil
end

local function moveNearProxySpawn(root, q)
    if not root or not root.Parent or not q then
        return false
    end

    local staticArea =
        MOB_SPAWNS[q.Mob]

    local targetCFrame =
        staticArea

    -- Proxy is allowed only if it is reasonably close
    -- to the known static spawn. This rejects template models at 0,0,0.
    local proxyRoot =
        findReplicatedMobRoot(q.Mob)

    if proxyRoot and staticArea then
        local proxyDistanceToStatic =
            (
                proxyRoot.Position
                - staticArea.Position
            ).Magnitude

        if proxyDistanceToStatic <= 650 then
            targetCFrame =
                proxyRoot.CFrame
        end

    elseif proxyRoot and not staticArea then
        targetCFrame =
            proxyRoot.CFrame
    end

    if not targetCFrame then
        return false
    end

    local targetPosition =
        targetCFrame.Position
        + Vector3.new(
            0,
            math.max(FarmState.HoverHeight, 20),
            0
        )

    local distance =
        (root.Position - targetPosition).Magnitude

    if distance > 22 and distance < 260 then
        pcall(function()
            root.CFrame =
                flatFacingCFrame(
                    targetPosition,
                    root.CFrame
                )

            root.AssemblyLinearVelocity =
                Vector3.zero
            root.AssemblyAngularVelocity =
                Vector3.zero
        end)
    end

    return true
end

local function travelToMobArea(root, humanoid, q)
    if not root or not root.Parent or not q then
        return false
    end

    local area = getMobAreaCFrame(q, root.Position)
    if not area then
        return false
    end

    -- Đang bay đúng bãi thì không tạo tween mới.
    if TravelState.Active
        and TravelState.TargetKey == q.Mob then
        return false
    end

    trySpecialEntrance(q)
    task.wait(0.10)

    if not root.Parent then
        return false
    end

    local finalPosition =
        area.Position
        + Vector3.new(
            0,
            math.max(FarmState.HoverHeight, 24),
            0
        )

    local finalCFrame =
        flatFacingCFrame(finalPosition, root.CFrame)

    if (root.Position - finalPosition).Magnitude <= 90 then
        return true
    end

    cancelTravel()
    TravelState.Token += 1

    local myToken = TravelState.Token

    TravelState.Active = true
    TravelState.MovementLock = true
    TravelState.TargetKey = q.Mob
    TravelState.TargetPosition = finalPosition
    TravelState.Humanoid = humanoid

    pcall(function()
        if humanoid then
            humanoid.Sit = false
            humanoid.AutoRotate = false
        end
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)

    task.spawn(function()
        local startPosition = root.Position
        local horizontalDistance =
            (
                Vector2.new(startPosition.X, startPosition.Z)
                - Vector2.new(finalPosition.X, finalPosition.Z)
            ).Magnitude

        local useSafeRoute = horizontalDistance > 300
        local ok = true

        if useSafeRoute then
            local safeY = math.max(
                TravelState.SafeBaseY,
                startPosition.Y + TravelState.SafeClearance,
                finalPosition.Y + TravelState.SafeClearance
            )

            -- 1) ASCEND
            local ascendPosition = Vector3.new(
                startPosition.X,
                safeY,
                startPosition.Z
            )

            ok = tweenTravelPhase(
                root,
                flatFacingCFrame(ascendPosition, root.CFrame),
                TravelState.VerticalSpeed,
                myToken
            )

            -- 2) CRUISE theo đoạn ngắn
            if ok then
                ok = travelHorizontalSegments(
                    root,
                    finalPosition,
                    safeY,
                    myToken
                )
            end

            -- 3) DESCEND
            if ok then
                ok = tweenTravelPhase(
                    root,
                    finalCFrame,
                    TravelState.VerticalSpeed,
                    myToken
                )
            end
        else
            ok = tweenTravelPhase(
                root,
                finalCFrame,
                TravelState.Speed,
                myToken
            )
        end

        if myToken ~= TravelState.Token then
            return
        end

        if ok then
            -- Giữ vị trí B ngắn để Farm/Quest không giành CFrame ngay khi vừa tới.
            local holdUntil = os.clock() + TravelState.ArrivalHold

            while os.clock() < holdUntil
                and myToken == TravelState.Token
                and root.Parent do
                pcall(function()
                    root.CFrame = finalCFrame
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                end)
                task.wait(0.06)
            end
        end

        if myToken ~= TravelState.Token then
            return
        end

        TravelState.Tween = nil
        TravelState.Active = false
        TravelState.MovementLock = false

        if ok then
            TravelState.ArrivedAt = os.clock()
            TravelState.LastArrivalKey = q.Mob
        end

        pcall(function()
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            if humanoid and humanoid.Parent then
                humanoid.AutoRotate = true
                humanoid.Sit = false
            end
        end)
    end)

    return false
end

local function getQuestTitle()
    local quest = getQuestGui()
    if not quest then
        return ""
    end

    local texts = {}

    -- Ưu tiên path cũ nếu vẫn còn.
    pcall(function()
        local direct = quest.Container.QuestTitle.Title.Text
        if type(direct) == "string" and direct ~= "" then
            table.insert(texts, direct)
        end
    end)

    -- Fallback mạnh hơn: gom text trong toàn bộ Quest GUI.
    -- Blox Fruits đổi layout vẫn còn bắt được tên mob.
    for _, obj in ipairs(quest:GetDescendants()) do
        if obj:IsA("TextLabel")
            or obj:IsA("TextButton")
            or obj:IsA("TextBox") then

            local ok, value = pcall(function()
                return obj.Text
            end)

            if ok
                and type(value) == "string"
                and value ~= "" then

                table.insert(texts, value)
            end
        end
    end

    return table.concat(texts, " | ")
end

local function getQuestEnemies(fromPosition, wantedName, maxTargets)
    local enemiesFolder = workspace:FindFirstChild("Enemies")
    if not enemiesFolder then
        return {}
    end

    local wanted = wantedName and string.lower(wantedName) or nil
    local found = {}

    for _, enemy in ipairs(enemiesFolder:GetChildren()) do
        if enemy:IsA("Model") then
            local humanoid = enemy:FindFirstChildOfClass("Humanoid")
            local root =
                enemy:FindFirstChild("HumanoidRootPart")
                or enemy:FindFirstChild("Torso")

            if humanoid and root and humanoid.Health > 0 then
                local nameOK = true

                if wanted then
                    nameOK = string.find(
                        string.lower(enemy.Name),
                        wanted,
                        1,
                        true
                    ) ~= nil
                end

                if nameOK then
                    table.insert(found, {
                        Enemy = enemy,
                        Humanoid = humanoid,
                        Root = root,
                        Distance = (root.Position - fromPosition).Magnitude,
                    })
                end
            end
        end
    end

    table.sort(found, function(a, b)
        return a.Distance < b.Distance
    end)

    local limit = math.min(maxTargets or #found, #found)
    local result = {}

    for i = 1, limit do
        result[i] = found[i]
    end

    return result
end

--=========================================================
-- STABLE MAGNET V1.10.2
--
-- Lỗi cũ:
--   anchor = enemyRoot.CFrame mỗi vòng.
-- Khi nhiều NPC chồng vật lý, root của con chính bị đẩy lên.
-- Player lại lấy root đó + HoverHeight => càng lúc càng bay cao.
--
-- Fix:
--   * Chốt 1 anchor cố định cho mỗi quest/cụm.
--   * Tắt collision của TOÀN BỘ BasePart NPC.
--   * Không ChangeState(Physics).
--   * Khôi phục WalkSpeed/JumpPower/AutoRotate khi reset.
--=========================================================

local MagnetState = {
    QuestKey = nil,
    AnchorCFrame = nil,
    AnchorY = nil,

    -- Multi-camp support.
    CampSwitchDistance = 135,
    CampKeepRadius = 105,
    LastCampSwitch = 0,
    CampSwitchCooldown = 0.45,
    CampIndex = 1,

    ModifiedHumanoids = setmetatable({}, {__mode = "k"}),
}

local function restoreMagnetHumanoids()
    for humanoid, original in pairs(MagnetState.ModifiedHumanoids) do
        if humanoid and humanoid.Parent then
            pcall(function()
                humanoid.WalkSpeed = original.WalkSpeed
                humanoid.JumpPower = original.JumpPower
                humanoid.AutoRotate = original.AutoRotate
            end)
        end
    end

    MagnetState.ModifiedHumanoids =
        setmetatable({}, {__mode = "k"})
end

local function resetMagnet()
    restoreMagnetHumanoids()

    MagnetState.QuestKey = nil
    MagnetState.AnchorCFrame = nil
    MagnetState.AnchorY = nil
    MagnetState.LastCampSwitch = 0
    MagnetState.CampIndex = 1
end

    else

    end

    return CFrame.lookAt(
        position,
        position + flat
    )
end

    if area then

    end

    local primary = targets and targets[1]
    local root = primary and primary.Root

    if root and root.Parent then
        return root.Position.Y
    end

    return nil
end

local function horizontalDistance(a, b)
    return (
        Vector2.new(a.X, a.Z)
        - Vector2.new(b.X, b.Z)
    ).Magnitude
end

local function targetNearCurrentCamp(targets)
    if not MagnetState.AnchorCFrame
        or type(targets) ~= "table" then

        return false
    end

    local anchorPos =
        MagnetState.AnchorCFrame.Position

    for _, data in ipairs(targets) do
        local root = data.Root

        if root and root.Parent then
            if horizontalDistance(
                root.Position,
                anchorPos
            ) <= MagnetState.CampKeepRadius then

                return true
            end
        end
    end

    return false
end

local function setCampAnchor(root)
    if not root or not root.Parent then
        return nil
    end

    -- Lock Y ONCE per camp.
    -- This keeps the anti-sky-drift fix while allowing camps
    -- with different heights to work correctly.
    local fixedY = root.Position.Y

    MagnetState.AnchorY = fixedY

    local fixedPosition =
        Vector3.new(
            root.Position.X,
            fixedY,
            root.Position.Z
        )

    MagnetState.AnchorCFrame =
        makeFlatCFrame(
            fixedPosition,
            root.CFrame
        )

    MagnetState.LastCampSwitch =
        os.clock()

    return MagnetState.AnchorCFrame
end

local function getStableMagnetAnchor(q, targets)
    local qKey = getQuestKey(q)

    if MagnetState.QuestKey ~= qKey then
        resetMagnet()
        MagnetState.QuestKey = qKey
    end

    local primary =
        targets and targets[1]

    local root =
        primary and primary.Root

    if not root or not root.Parent then
        return MagnetState.AnchorCFrame
    end

    -- First camp.
    if not MagnetState.AnchorCFrame then
        return setCampAnchor(root)
    end

    local currentAnchor =
        MagnetState.AnchorCFrame.Position

    local primaryDistance =
        horizontalDistance(
            root.Position,
            currentAnchor
        )

    local currentCampStillAlive =
        targetNearCurrentCamp(targets)

    -- IMPORTANT MULTI-CAMP FIX:
    -- If camp A is cleared and the nearest live mob is in camp B,
    -- release the old quest anchor and lock onto camp B.
    if not currentCampStillAlive
        and primaryDistance
            >= MagnetState.CampSwitchDistance
        and os.clock()
            - MagnetState.LastCampSwitch
            >= MagnetState.CampSwitchCooldown then

        MagnetState.CampIndex += 1

        return setCampAnchor(root)
    end

    return MagnetState.AnchorCFrame
end

local function clampAnchorToFixedY(anchorCFrame)
    if not anchorCFrame or not MagnetState.AnchorY then
        return anchorCFrame
    end

    local p = anchorCFrame.Position

    return makeFlatCFrame(
        Vector3.new(
            p.X,
            MagnetState.AnchorY,
            p.Z
        ),
        anchorCFrame
    )
end

local function magnetQuestEnemies(targets, anchorCFrame)
    if not FarmState.MagnetEnabled
        or not anchorCFrame
        or type(targets) ~= "table" then

        return
    end

    tryExpandSimulationRadius()

    anchorCFrame = clampAnchorToFixedY(anchorCFrame)

    if not anchorCFrame then
        return
    end

    local fixedY = MagnetState.AnchorY
        or anchorCFrame.Position.Y

    for index, data in ipairs(targets) do
        local enemy = data.Enemy
        local humanoid = data.Humanoid
        local root = data.Root

        if enemy and enemy.Parent
            and humanoid and humanoid.Health > 0
            and root and root.Parent then

            local horizontalDistance =
                (
                    Vector2.new(root.Position.X, root.Position.Z)
                    - Vector2.new(
                        anchorCFrame.Position.X,
                        anchorCFrame.Position.Z
                    )
                ).Magnitude

            if horizontalDistance <= FarmState.MagnetRadius then
                pcall(function()
                    if not MagnetState.ModifiedHumanoids[humanoid] then
                        MagnetState.ModifiedHumanoids[humanoid] = {
                            WalkSpeed = humanoid.WalkSpeed,
                            JumpPower = humanoid.JumpPower,
                            AutoRotate = humanoid.AutoRotate,
                        }
                    end

                    humanoid.WalkSpeed = 0
                    humanoid.JumpPower = 0
                    humanoid.AutoRotate = false
                    humanoid.Sit = false

                    for _, part in ipairs(enemy:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.CanCollide = false
                            part.AssemblyLinearVelocity = Vector3.zero
                            part.AssemblyAngularVelocity = Vector3.zero
                        end
                    end

                    local ringIndex = (index - 1) % 8
                    local angle = math.rad(ringIndex * 45)
                    local radius = index == 1 and 0 or 1.65

                    local fixedPosition = Vector3.new(
                        anchorCFrame.Position.X + math.cos(angle) * radius,
                        fixedY,
                        anchorCFrame.Position.Z + math.sin(angle) * radius
                    )

                    root.CFrame =
                        makeFlatCFrame(
                            fixedPosition,
                            anchorCFrame
                        )

                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                end)
            end
        end
    end
end

local function toolPriority(tool)
    if not tool:IsA("Tool") then
        return -1
    end

    local tip = string.lower(tostring(tool.ToolTip or ""))

    if tip == "melee" or tip:find("melee", 1, true) then
        return 300
    end

    if tip == "sword" or tip:find("sword", 1, true) then
        return 200
    end

    return 100
end

local function findBestTool(character)
    local equipped = character:FindFirstChildOfClass("Tool")
    if equipped then
        return equipped
    end

    local backpack = Player:FindFirstChildOfClass("Backpack")
        or Player:FindFirstChild("Backpack")

    if not backpack then
        return nil
    end

    local bestTool = nil
    local bestPriority = -1

    for _, child in ipairs(backpack:GetChildren()) do
        local priority = toolPriority(child)

        if priority > bestPriority then
            bestPriority = priority
            bestTool = child
        end
    end

    return bestTool
end

local function equipBestTool(character, humanoid)
    local tool = findBestTool(character)

    if tool and tool.Parent ~= character then
        pcall(function()
            humanoid:EquipTool(tool)
        end)
    end

    if tool and tool.Parent == character then
        return tool
    end

    return character:FindFirstChildOfClass("Tool")
end

local MultiAttack = {
    Ready = false,
    Error = nil,
    RegisterAttack = nil,
    RegisterHit = nil,
    LastAttack = 0,

    -- Giá trị thực sẽ được startFarm() ép theo FastRemoteDelay.
    Delay = 0.070,
}

local function initMultiAttack()
    if MultiAttack.Ready then
        return true
    end

    local modules = ReplicatedStorage:FindFirstChild("Modules")
    local net = modules and modules:FindFirstChild("Net")

    if not net then
        MultiAttack.Error = "NO_MODULES_NET"
        return false
    end

    local registerAttack = net:FindFirstChild("RE/RegisterAttack")
    local registerHit = net:FindFirstChild("RE/RegisterHit")

    if not registerAttack then
        MultiAttack.Error = "NO_RE_REGISTER_ATTACK"
        return false
    end

    if not registerHit then
        MultiAttack.Error = "NO_RE_REGISTER_HIT"
        return false
    end

    MultiAttack.RegisterAttack = registerAttack
    MultiAttack.RegisterHit = registerHit
    MultiAttack.Ready = true
    MultiAttack.Error = nil

    return true
end

local function getPreferredHitPart(enemy)
    if not enemy or not enemy.Parent then
        return nil
    end

    local preferred = {
        "Head",
        "HumanoidRootPart",
        "UpperTorso",
        "Torso",
        "RightLowerLeg",
        "LeftLowerLeg",
    }

    for _, name in ipairs(preferred) do
        local part = enemy:FindFirstChild(name)
        if part and part:IsA("BasePart") then
            return part
        end
    end

    return enemy:FindFirstChildWhichIsA("BasePart", true)
end

--=========================================================
-- ATTACK V1.10.5
--
-- Combat/Fighting Style hiện tại có thể chứa LeftClickRemote.
-- Với loại Tool này, Blox Fruits xử lý đòn đánh qua:
--     Tool.LeftClickRemote:FireServer(direction, 1)
--
-- Nếu Tool không có LeftClickRemote:
--     fallback RE/RegisterAttack + RE/RegisterHit.
--
-- Không VirtualUser / không auto-click màn hình.
--=========================================================

--=========================================================
-- ATTACK V1.10.6 - ADAPTIVE COMBAT
--
-- Tự kiểm tra damage thật:
--   LEFTCLICK_REMOTE không làm máu mob giảm sau vài lần
--      -> tự chuyển NET_REGISTER_HIT.
--   NET không gây damage
--      -> thử lại LEFTCLICK_REMOTE.
--
-- Không VirtualUser / không auto-click màn hình.
--=========================================================

local AdaptiveCombat = {
    Pending = nil,

    LeftFail = 0,
    NetFail = 0,

    Preferred = "AUTO",
    LastSwitch = 0,
    SwitchCooldown = 1.2,

    DamageCheckDelay = 0.16,
    FailThreshold = 3,
}

local function getLeftClickRemote(tool)
    if not tool then
        return nil
    end

    local direct = tool:FindFirstChild("LeftClickRemote")

    if direct and direct:IsA("RemoteEvent") then
        return direct
    end

    for _, obj in ipairs(tool:GetDescendants()) do
        if obj.Name == "LeftClickRemote"
            and obj:IsA("RemoteEvent") then

            return obj
        end
    end

    return nil
end

local function collectAttackTargets(targetData, playerRoot)
    local valid = {}
    local others = {}
    local basePart = nil

    for _, data in ipairs(targetData) do
        local enemy = data.Enemy
        local humanoid = data.Humanoid

        if enemy and enemy.Parent
            and humanoid and humanoid.Health > 0 then

            local part = getPreferredHitPart(enemy)
            local root =
                enemy:FindFirstChild("HumanoidRootPart")
                or part

            if part and root then
                local distance =
                    (part.Position - playerRoot.Position).Magnitude

                if distance <= FarmState.HitDistance then
                    basePart = basePart or part

                    table.insert(valid, {
                        Enemy = enemy,
                        Humanoid = humanoid,
                        Part = part,
                        Root = root,
                    })

                    table.insert(others, {
                        enemy,
                        part,
                    })
                end
            end
        end
    end

    return valid, basePart, others
end

local function snapshotHealth(validTargets, backend)
    -- Fast Attack có thể đánh nhiều lần trước khi đủ thời gian kiểm tra damage.
    -- Không overwrite snapshot cũ; chờ evaluatePendingDamage() xử lý xong.
    if AdaptiveCombat.Pending then
        return
    end

    local snapshot = {}

    for _, data in ipairs(validTargets) do
        local humanoid = data.Humanoid

        if humanoid and humanoid.Parent then
            table.insert(snapshot, {
                Humanoid = humanoid,
                Health = humanoid.Health,
            })
        end
    end

    AdaptiveCombat.Pending = {
        Backend = backend,
        Time = os.clock(),
        Targets = snapshot,
    }
end

local function evaluatePendingDamage()
    local pending = AdaptiveCombat.Pending

    if not pending then
        return nil
    end

    if os.clock() - pending.Time
        < AdaptiveCombat.DamageCheckDelay then

        return nil
    end

    AdaptiveCombat.Pending = nil

    local damaged = false

    for _, item in ipairs(pending.Targets) do
        local humanoid = item.Humanoid

        if humanoid and humanoid.Parent then
            if humanoid.Health < item.Health
                or humanoid.Health <= 0 then

                damaged = true
                break
            end
        end
    end

    if pending.Backend == "LEFTCLICK_REMOTE" then
        if damaged then
            AdaptiveCombat.LeftFail = 0
            AdaptiveCombat.Preferred = "LEFT"
        else
            AdaptiveCombat.LeftFail += 1

            if AdaptiveCombat.LeftFail
                >= AdaptiveCombat.FailThreshold then

                AdaptiveCombat.Preferred = "NET"
                AdaptiveCombat.LastSwitch = os.clock()
                AdaptiveCombat.LeftFail = 0
            end
        end

    elseif pending.Backend == "MULTI_NET_HIT" then
        if damaged then
            AdaptiveCombat.NetFail = 0
            AdaptiveCombat.Preferred = "NET"
        else
            AdaptiveCombat.NetFail += 1

            if AdaptiveCombat.NetFail
                >= AdaptiveCombat.FailThreshold then

                AdaptiveCombat.Preferred = "LEFT"
                AdaptiveCombat.LastSwitch = os.clock()
                AdaptiveCombat.NetFail = 0
            end
        end
    end

    return damaged
end

local function attackWithLeftClickRemote(
    leftClickRemote,
    validTargets,
    playerRoot
)
    if not leftClickRemote
        or #validTargets == 0 then

        return false, 0
    end

    local character = Player.Character

    if not character then
        return false, 0
    end

    local origin = playerRoot.Position

    pcall(function()
        origin = character:GetPivot().Position
    end)

    local hitCount = 0

    -- Giới hạn remote spam.
    local limit = math.min(#validTargets, 6)

    for i = 1, limit do
        local data = validTargets[i]
        local targetRoot = data.Root

        if targetRoot and targetRoot.Parent then
            local delta =
                targetRoot.Position - origin

            if delta.Magnitude > 0.01 then
                local ok = pcall(function()
                    leftClickRemote:FireServer(
                        delta.Unit,
                        1
                    )
                end)

                if ok then
                    hitCount += 1
                end
            end
        end
    end

    return hitCount > 0, hitCount
end

local function attackWithNetRemotes(
    basePart,
    others
)
    if not initMultiAttack() then
        return false,
            MultiAttack.Error or "NET_INIT_FAILED",
            0
    end

    if not basePart or #others == 0 then
        return false, "NO_TARGETS_IN_RANGE", 0
    end

    local attackOK = pcall(function()
        -- 0.5 ổn định hơn 0 với RegisterAttack hiện tại.
        MultiAttack.RegisterAttack:FireServer(0.5)
    end)

    if not attackOK then
        return false, "REGISTER_ATTACK_FAILED", 0
    end

    task.wait()

    local hitOK = pcall(function()
        MultiAttack.RegisterHit:FireServer(
            basePart,
            others
        )
    end)

    if not hitOK then
        return false, "REGISTER_HIT_FAILED", 0
    end

    return true, "MULTI_NET_HIT", #others
end

local function useLeftBackend(
    leftClickRemote,
    validTargets,
    playerRoot
)
    if not leftClickRemote then
        return false, "NO_LEFTCLICK_REMOTE", 0
    end

    local ok, count =
        attackWithLeftClickRemote(
            leftClickRemote,
            validTargets,
            playerRoot
        )

    if ok then
        snapshotHealth(
            validTargets,
            "LEFTCLICK_REMOTE"
        )

        return true, "LEFTCLICK_REMOTE", count
    end

    return false, "LEFTCLICK_FAILED", 0
end

local function useNetBackend(
    validTargets,
    basePart,
    others
)
    local ok, backend, count =
        attackWithNetRemotes(
            basePart,
            others
        )

    if ok then
        snapshotHealth(
            validTargets,
            "MULTI_NET_HIT"
        )
    end

    return ok, backend, count
end

local function multiMobAttack(tool, targetData, playerRoot)
    if not tool or not tool.Parent then
        return false, "NO_TOOL", 0
    end

    if not playerRoot then
        return false, "NO_PLAYER_ROOT", 0
    end

    -- Đọc kết quả damage của lần đánh trước.
    evaluatePendingDamage()

    local now = os.clock()

    if now - MultiAttack.LastAttack
        < MultiAttack.Delay then

        return false, "ATTACK_COOLDOWN", 0
    end

    local validTargets, basePart, others =
        collectAttackTargets(
            targetData,
            playerRoot
        )

    if #validTargets == 0 then
        return false, "NO_TARGETS_IN_RANGE", 0
    end

    MultiAttack.LastAttack = now

    local leftClickRemote =
        getLeftClickRemote(tool)

    -- Nếu vừa switch backend, giữ lựa chọn đó một lúc
    -- để có đủ thời gian kiểm tra damage thật.
    local forced =
        os.clock() - AdaptiveCombat.LastSwitch
        < AdaptiveCombat.SwitchCooldown

    --=====================================================
    -- AUTO / LEFT preference
    --=====================================================

    if leftClickRemote
        and AdaptiveCombat.Preferred ~= "NET" then

        local ok, backend, count =
            useLeftBackend(
                leftClickRemote,
                validTargets,
                playerRoot
            )

        if ok then
            return true, backend, count
        end
    end

    --=====================================================
    -- NET preference / fallback
    --=====================================================

    local netOK, netBackend, netCount =
        useNetBackend(
            validTargets,
            basePart,
            others
        )

    if netOK then
        return true, netBackend, netCount
    end

    --=====================================================
    -- NET fail -> thử LeftClick lần cuối nếu có.
    --=====================================================

    if leftClickRemote then
        local ok, backend, count =
            useLeftBackend(
                leftClickRemote,
                validTargets,
                playerRoot
            )

        if ok then
            return true, backend, count
        end
    end

    return false,
        netBackend or "ALL_ATTACK_BACKENDS_FAILED",
        0
end

local function safeAttack(tool, targets, playerRoot)
    return multiMobAttack(
        tool,
        targets,
        playerRoot
    )
end

local function moveToQuest(root, q)
    if not root or not q then
        return false
    end

    pcall(function()
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        root.CFrame = q.CFrame * CFrame.new(0, 2.5, 0)
    end)

    return (root.Position - q.CFrame.Position).Magnitude <= 20
end

local function requestQuest(q, root)
    if not q or not root then
        return false
    end

    -- Travel đang sở hữu CFrame: Quest không được kéo ngược về NPC.
    if TravelState and TravelState.MovementLock then
        return false
    end

    if os.clock() - FarmState.LastQuestAttempt < FarmState.QuestRetryDelay then
        return false
    end

    FarmState.LastQuestAttempt = os.clock()

    local level = getPlayerLevel()
    tryTravelToSubmerged(level)

    moveToQuest(root, q)
    task.wait(0.22)

    local remote = getCommRemote()
    if not remote then
        return false
    end

    local ok = pcall(function()
        remote:InvokeServer("StartQuest", q.Quest, q.Index)
    end)

    task.wait(0.28)

    if ok then
        FarmState.ActiveQuestKey = getQuestKey(q)
    end

    return ok
end

local function abandonQuest()
    FarmState.ActiveQuestKey = nil

    local remote = getCommRemote()
    if not remote then
        return
    end

    pcall(function()
        remote:InvokeServer("AbandonQuest")
    end)
end

local function startFarm()
    if FarmState.Enabled then
        return
    end

    FarmState.Enabled = true
    FarmState.FastAttackActive = true
    FarmState.Token += 1
    FarmState.LastQuestAttempt = 0
    FarmState.NoMobSince = 0
    FarmState.LastNoMobQuest = nil

    -- Bật Farm là Fast Attack bật ngay, không cần toggle riêng.
    pcall(function()
        MultiAttack.Delay = FarmState.FastRemoteDelay
        MultiAttack.LastAttack = 0

        AdaptiveCombat.Pending = nil
        AdaptiveCombat.LeftFail = 0
        AdaptiveCombat.NetFail = 0
        AdaptiveCombat.Preferred = "AUTO"
    end)

    local myToken = FarmState.Token
    local lastAttack = 0

    updateToggleVisual()

    if GameReady then
        FarmStatus.Text =
            "Farm ON • Fast Attack ON • Đang khởi động..."
    else
        FarmStatus.Text =
            "Farm ON • chờ game load..."
    end

    -- Scope sanity check để tránh lỗi kiểu V1.10.2 quay lại.
    if type(getQuestKey) ~= "function" then
        FarmStatus.Text = "Farm lỗi: getQuestKey scope"
        FarmState.Enabled = false
        updateToggleVisual()
        return
    end

    if type(resetMagnet) ~= "function"
        or type(getStableMagnetAnchor) ~= "function" then

        FarmStatus.Text = "Farm lỗi: Magnet engine scope"
        FarmState.Enabled = false
        updateToggleVisual()
        return
    end

    task.spawn(function()
        local function farmLoop()
            while FarmState.Enabled
                and FarmState.Token == myToken
                and ScreenGui.Parent do

                if not GameReady then
                    FarmStatus.Text =
                        "Farm ON • đang chờ CommF_/Backpack..."

                    task.wait(0.20)
                    continue
                end

                local character, humanoid, playerRoot = getCharacter()

                -- HARD TRAVEL LOCK: trong lúc bay, Farm/Quest không được chạm CFrame.
                if TravelState and TravelState.MovementLock then
                    task.wait(0.06)
                    continue
                end

                -- Auto nhặt fruit tạm giữ quyền di chuyển.
                -- Tránh Farm kéo nhân vật ngược lại trong lúc pickup.
                if FruitState and FruitState.Busy then
                    task.wait(0.08)
                    continue
                end

                if not character then
                    FarmStatus.Text = "Trạng thái: Đợi nhân vật..."
                    task.wait(0.25)
                    continue
                end

                local level = getPlayerLevel()
                local q = resolveQuest(level)

                if q then
                    local resolvedKey = getQuestKey(q)

                    -- Level vừa vượt mốc quest:
                    -- reset cụm cũ + bỏ quest cũ ngay.
                    if FarmState.LastResolvedQuestKey
                        and FarmState.LastResolvedQuestKey ~= resolvedKey then

                        cancelTravel()
                        resetMagnet()
                        FarmState.ActiveQuestKey = nil

                        if isQuestVisible() then
                            FarmStatus.Text =
                                "Đổi quest theo level → " .. q.Mob

                            abandonQuest()
                            task.wait(0.30)
                            FarmState.LastResolvedQuestKey = resolvedKey
                            continue
                        end
                    end

                    FarmState.LastResolvedQuestKey = resolvedKey
                end

                if not q then
                    FarmStatus.Text =
                        "Không có dữ liệu quest cho Lv." .. tostring(level)

                    task.wait(0.5)
                    continue
                end

                --=================================================
                -- 1) AUTO QUEST
                --=================================================

                if not isQuestVisible() then
                    local inArrivalGrace =
                        TravelState.LastArrivalKey == q.Mob
                        and (
                            os.clock()
                            - (TravelState.ArrivedAt or 0)
                        ) < TravelState.ArrivalGrace

                    if inArrivalGrace then
                        -- Sau khi vừa sang đảo mới, GUI/streaming có thể
                        -- trễ vài giây. Không teleport ngược về quest NPC.
                        FarmStatus.Text =
                            "Đã tới " .. q.Mob ..
                            " | chờ quest/mob sync..."

                        moveNearProxySpawn(
                            playerRoot,
                            q
                        )

                        task.wait(0.12)
                        continue
                    end

                    cancelTravel()
                    resetMagnet()

                    FarmStatus.Text =
                        "Nhận quest Lv." .. tostring(level) ..
                        ": " .. q.Mob

                    requestQuest(q, playerRoot)

                    task.wait(0.18)
                    continue
                end

                if not questMatches(q) then
                    cancelTravel()
                    resetMagnet()

                    FarmStatus.Text =
                        "Quest sai → nhận lại: " .. q.Mob

                    abandonQuest()

                    task.wait(0.35)
                    continue
                end

                --=================================================
                -- 2) TÌM MOB ĐÚNG QUEST
                --=================================================

                local targets = getQuestEnemies(
                    playerRoot.Position,
                    q.Mob,
                    FarmState.MaxTargets
                )

                --=================================================
                -- 3) MOB CHƯA LOAD -> BAY TỚI BÃI MOB
                --=================================================

                if #targets == 0 then
                    if FarmState.LastNoMobQuest ~= q.Mob then
                        FarmState.LastNoMobQuest = q.Mob
                        FarmState.NoMobSince = os.clock()
                    elseif FarmState.NoMobSince == 0 then
                        FarmState.NoMobSince = os.clock()
                    end

                    local mobArea =
                        getMobAreaCFrame(
                            q,
                            playerRoot.Position
                        )

                    if mobArea then
                        local distance =
                            (
                                playerRoot.Position
                                - mobArea.Position
                            ).Magnitude

                        if distance > 105 then
                            FarmStatus.Text =
                                "Bay tới " .. q.Mob ..
                                " | " ..
                                tostring(
                                    math.floor(distance)
                                ) ..
                                " studs"

                            travelToMobArea(
                                playerRoot,
                                humanoid,
                                q
                            )
                        else
                            cancelTravel()

                            moveNearProxySpawn(
                                playerRoot,
                                q
                            )

                            local waitTime =
                                os.clock()
                                - FarmState.NoMobSince

                            FarmStatus.Text =
                                "Đã tới bãi " ..
                                q.Mob ..
                                " | chờ mob " ..
                                tostring(
                                    math.floor(waitTime)
                                ) ..
                                "s"

                            -- Sau 5s vẫn chưa thấy mob:
                            -- nudge quanh CFrameMon để ép streaming/spawn vùng.
                            if waitTime >= 5 then
                                local phase =
                                    math.floor(waitTime) % 4

                                local offsets = {
                                    Vector3.new(35, 20, 0),
                                    Vector3.new(-35, 20, 0),
                                    Vector3.new(0, 20, 35),
                                    Vector3.new(0, 20, -35),
                                }

                                local nudge =
                                    mobArea.Position
                                    + offsets[phase + 1]

                                pcall(function()
                                    playerRoot.CFrame =
                                        flatFacingCFrame(
                                            nudge,
                                            playerRoot.CFrame
                                        )
                                end)
                            end
                        end
                    else
                        FarmStatus.Text =
                            "Thiếu CFrame bãi: " ..
                            q.Mob
                    end

                    task.wait(0.12)
                    continue
                end

                FarmState.NoMobSince = 0
                FarmState.LastNoMobQuest = nil

                --=================================================
                -- 4) MOB ĐÃ LOAD -> DỪNG BAY, GOM QUÁI
                --=================================================

                cancelTravel()

                local primary = targets[1]

                if not primary then
                    task.wait(0.08)
                    continue
                end

                local enemy = primary.Enemy
                local enemyHumanoid = primary.Humanoid
                local enemyRoot = primary.Root

                if not enemy
                    or not enemy.Parent
                    or not enemyHumanoid
                    or enemyHumanoid.Health <= 0
                    or not enemyRoot
                    or not enemyRoot.Parent then

                    task.wait(0.08)
                    continue
                end

                local magnetCFrame =
                    getStableMagnetAnchor(q, targets)

                if not magnetCFrame then
                    FarmStatus.Text =
                        "Đang khóa điểm farm: " .. q.Mob

                    task.wait(0.08)
                    continue
                end

                magnetQuestEnemies(
                    targets,
                    magnetCFrame
                )

                --=================================================
                -- 5) ĐỨNG TRÊN CỤM QUÁI
                --=================================================

                --=================================================
                -- UPRIGHT HOVER FIX V1.10.1
                --
                -- Không dùng lookAt() nhìn thẳng xuống mob nữa,
                -- vì nó làm cả avatar bị xoay nằm ngang.
                -- Chỉ lấy hướng ngang XZ để nhân vật luôn đứng thẳng.
                --=================================================

                magnetCFrame =
                    clampAnchorToFixedY(magnetCFrame)

                local mobPosition = Vector3.new(
                    magnetCFrame.Position.X,
                    MagnetState.AnchorY
                        or magnetCFrame.Position.Y,
                    magnetCFrame.Position.Z
                )

                local flatLook = Vector3.new(
                    magnetCFrame.LookVector.X,
                    0,
                    magnetCFrame.LookVector.Z
                )

                if flatLook.Magnitude < 0.01 then
                    flatLook = Vector3.new(0, 0, -1)
                else
                    flatLook = flatLook.Unit
                end

                -- Đứng cao + lệch nhẹ ra sau mob.
                local desiredY =
                    mobPosition.Y + FarmState.HoverHeight

                local maxAllowedY =
                    mobPosition.Y + FarmState.MaxPlayerAboveAnchor

                desiredY = math.min(
                    desiredY,
                    maxAllowedY
                )

                local farmPosition =
                    Vector3.new(
                        mobPosition.X,
                        desiredY,
                        mobPosition.Z
                    )
                    - flatLook * FarmState.BackOffset

                -- Nhìn về phía mob NHƯNG giữ cùng độ cao Y.
                -- => chỉ quay yaw, không pitch xuống dưới.
                local flatTarget = Vector3.new(
                    mobPosition.X,
                    farmPosition.Y,
                    mobPosition.Z
                )

                local targetCFrame

                if (flatTarget - farmPosition).Magnitude > 0.05 then
                    targetCFrame = CFrame.lookAt(
                        farmPosition,
                        flatTarget
                    )
                else
                    targetCFrame = CFrame.new(
                        farmPosition,
                        farmPosition + flatLook
                    )
                end

                pcall(function()
                    if humanoid then
                        humanoid.AutoRotate = false
                        humanoid.Sit = false
                    end

                    -- Chỉ reset vận tốc nếu thực sự bị đẩy,
                    -- giảm kiểu rung do physics bị ép mỗi vòng.
                    if playerRoot.AssemblyLinearVelocity.Magnitude > 2 then
                        playerRoot.AssemblyLinearVelocity =
                            Vector3.zero
                    end

                    if playerRoot.AssemblyAngularVelocity.Magnitude > 0.5 then
                        playerRoot.AssemblyAngularVelocity =
                            Vector3.zero
                    end

                    -- Tránh spam lại CFrame nếu đã ở gần đúng vị trí.
                    local positionError =
                        (playerRoot.Position - farmPosition).Magnitude

                    local currentLook = Vector3.new(
                        playerRoot.CFrame.LookVector.X,
                        0,
                        playerRoot.CFrame.LookVector.Z
                    )

                    local wantedLook = Vector3.new(
                        targetCFrame.LookVector.X,
                        0,
                        targetCFrame.LookVector.Z
                    )

                    local rotationNeedsUpdate = true

                    if currentLook.Magnitude > 0.01
                        and wantedLook.Magnitude > 0.01 then

                        currentLook = currentLook.Unit
                        wantedLook = wantedLook.Unit

                        rotationNeedsUpdate =
                            currentLook:Dot(wantedLook) < 0.985
                    end

                    if positionError > 0.65
                        or rotationNeedsUpdate then

                        playerRoot.CFrame = targetCFrame
                    end
                end)

                --=================================================
                -- 6) EQUIP + MULTI HIT
                --=================================================

                pcall(function()
                    local maxY =
                        mobPosition.Y
                        + FarmState.MaxPlayerAboveAnchor

                    if playerRoot.Position.Y > maxY + 1 then
                        playerRoot.CFrame = targetCFrame
                        playerRoot.AssemblyLinearVelocity =
                            Vector3.zero
                        playerRoot.AssemblyAngularVelocity =
                            Vector3.zero
                    end
                end)

                local tool =
                    equipBestTool(character, humanoid)

                local now = os.clock()

                if tool
                    and FarmState.FastAttackActive
                    and (now - lastAttack)
                        >= FarmState.AttackDelay then

                    lastAttack = now

                    local ok, backend, hitCount =
                        safeAttack(
                            tool,
                            targets,
                            playerRoot
                        )

                    if ok then
                        FarmStatus.Text =
                            "FAST | Lv." .. tostring(level) ..
                            " | " .. q.Mob ..
                            " | Bãi " ..
                            tostring(MagnetState.CampIndex) ..
                            " | x" ..
                            tostring(hitCount or 0) ..
                            " | " .. tostring(backend)
                    elseif backend ~= "ATTACK_COOLDOWN" then
                        FarmStatus.Text =
                            "Combat: " .. tostring(backend) ..
                            " | Mob: " .. tostring(#targets)
                    end

                elseif not tool then
                    FarmStatus.Text =
                        "Không tìm thấy Tool để đánh"
                end

                task.wait(FarmState.UpdateDelay)
            end
        end

        -- Nếu có lỗi runtime, không để toggle chết im.
        local ok, err = xpcall(
            farmLoop,
            function(message)
                return tostring(message)
            end
        )

        if not ok then
            FarmStatus.Text =
                "Farm lỗi: " .. tostring(err)

            warn(
                "[AXIOM FARM ERROR]",
                tostring(err)
            )

            FarmState.Enabled = false
            FarmState.FastAttackActive = false
            FarmState.Token += 1
            cancelTravel()
            resetMagnet()

            pcall(function()
                local character = Player.Character
                local humanoid = character
                    and character:FindFirstChildOfClass("Humanoid")

                if humanoid then
                    humanoid.AutoRotate = true
                end
            end)

            updateToggleVisual()
        end
    end)
end

Toggle.MouseButton1Click:Connect(function()
    if FarmState.Enabled then
        stopFarm()
    else
        startFarm()
    end
end)

ScreenGui.Destroying:Connect(function()
    FarmState.Enabled = false
    FarmState.FastAttackActive = false
    FarmState.Token += 1
    FarmState.ActiveQuestKey = nil
    FarmState.LastResolvedQuestKey = nil
    cancelTravel()
    resetMagnet()
end)

--=========================================================
-- DRAG SYSTEM
--=========================================================

local dragging = false
local dragStart
local startPos

TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

TopBar.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then

        local delta = input.Position - dragStart

        Main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = false
    end
end)

--=========================================================
-- MINIMIZE SYSTEM

local minimized = false
local originalSize = Main.Size

Minimize.MouseButton1Click:Connect(function()

    if minimized then

        Content.Visible = false

        TweenService:Create(

        ):Play()

            {Size = originalSize}
        ):Play()

        task.delay(0.2, function()
            Sidebar.Visible = true
            Content.Visible = true
        end)
    end
end)

Close.MouseButton1Click:Connect(function()
    ScreenGui:Destroy()
end)


--=========================================================
-- UI LAYER NORMALIZER V1.22.2
-- Một số executor Android render child ZIndex thấp hơn parent background.
-- Chuẩn hóa toàn bộ nội dung tab sau khi tất cả panel đã được tạo.
--=========================================================

local function normalizePageLayers(page)
    if not page then
        return
    end

    pcall(function()
        page.Visible = page.Visible
        page.ZIndex = 2
    end)

    for _, obj in ipairs(page:GetDescendants()) do
        if obj:IsA("GuiObject") then
            pcall(function()
                if obj.ZIndex < 3 then
                    obj.ZIndex = 3
                end
            end)
        end
    end
end

for _, page in pairs(tabs) do
    normalizePageLayers(page)
end

-- Content chỉ làm nền, luôn thấp hơn page content.
pcall(function()
    Content.ZIndex = 1
end)

-- Sidebar/tab button vẫn nổi trên nền.
pcall(function()
    Sidebar.ZIndex = 2
end)

for _, button in pairs(tabButtons) do
    pcall(function()
        button.ZIndex = 3
    end)
end

-- Ép lại Farm sau khi normalize để tránh executor giữ state Visible cũ.
switchTab("Farm")

print("[AXIOM] UI LAYERS NORMALIZED")
